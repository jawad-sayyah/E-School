MiniShop works, but nothing stops `dispatch({ type: "ad" })`, a price stored as `"12"`, or `setQty(id)` with no quantity. Each of those is a bug you'd only find by clicking around. TypeScript finds them as you type.

Converting an existing app is a routine job in real teams. The recipe:

1. Rename files: `.js` → `.ts`, and files with JSX → `.tsx`. (In a Vite project, install `typescript`, add a `tsconfig.json` with `"strict": true`, and `npm run build` type-checks.)
2. Describe your data once, in `types.ts`: `Product`, `CartItem` and `CartAction`.
3. Type the boundaries: function parameters and return types, the context value, and props. Inside functions, inference does most of the work.
4. Fix what the compiler finds.

You'll do the two parts that need real thought, the reducer and the context, in the type-checked editor. Then you'll see the whole converted app running.

Replace the three `unknown` types with real ones, and type `addToCart` and `cartReducer` (parameters and return types). The hidden checks make sure a misspelled action, a missing quantity or a price written as a string are all compile errors.

```ts starter
type Product = unknown;
type CartItem = unknown;
type CartAction = unknown;

function addToCart(items, product) {
  const existing = items.find((item) => item.id === product.id);
  if (existing) {
    return items.map((item) => (item.id === product.id ? { ...item, qty: item.qty + 1 } : item));
  }
  return [...items, { ...product, qty: 1 }];
}

function cartReducer(items, action) {
  switch (action.type) {
    case "add":
      return addToCart(items, action.product);
    case "remove":
      return items.filter((item) => item.id !== action.id);
    case "setQty":
      if (action.qty < 1) return items.filter((item) => item.id !== action.id);
      return items.map((item) => (item.id === action.id ? { ...item, qty: action.qty } : item));
    case "clear":
      return [];
    default:
      throw new Error(`Unknown action: ${action.type}`);
  }
}
```

```ts solution
  id: number;
  name: string;
  price: number;
}

interface CartItem extends Product {
  qty: number;
}

type CartAction =
  | { type: "add"; product: Product }
  | { type: "remove"; id: number }
  | { type: "setQty"; id: number; qty: number }
  | { type: "clear" };

function addToCart(items: CartItem[], product: Product): CartItem[] {

  }
  return [...items, { ...product, qty: 1 }];
}

    case "remove":
      return items.filter((item) => item.id !== action.id);
    case "setQty":

    default: {

    }
  }
```

<HowItWorks title="Why `never` in the default case?">

Inside each `case`, TypeScript **narrows** `action` to the matching member of the union: in `case "setQty"` it knows `action.qty` exists. After the last `case`, every member has been handled, so in `default` the type of `action` is `never`, the type with no values.

`const unhandled: never = action` only compiles while that's true. Add a fifth action, `{ type: "applyCoupon"; code: string }`, without a `case`, and that line becomes an error that points straight at the reducer. The `throw` is still there for the run time, for actions that come from untyped places like old saved data or a browser extension.

</HowItWorks>

The context is untyped, so `createContext(null)` makes its value `null` forever and nothing compiles. Type the context value, the provider's `children` and `useCart`'s functions. (The data types and the reducer are declared at the top, as in your real files.)

```tsx starter
import { createContext, useContext, useReducer, type Dispatch, type ReactNode } from "react";

interface Product { id: number; name: string; price: number }
interface CartItem extends Product { qty: number }
type CartAction =
  | { type: "add"; product: Product }
  | { type: "remove"; id: number }
  | { type: "setQty"; id: number; qty: number }
  | { type: "clear" };
declare function cartReducer(items: CartItem[], action: CartAction): CartItem[];
declare function loadCart(): CartItem[];

const CartContext = createContext(null);

export function CartProvider({ children }) {
  const [items, dispatch] = useReducer(cartReducer, undefined, loadCart);
  return <CartContext value={{ items, dispatch }}>{children}</CartContext>;
}

export function useCart() {
  const context = useContext(CartContext);
  if (!context) throw new Error("useCart must be used inside <CartProvider>");
  return {
    items: context.items,
    add: (product) => context.dispatch({ type: "add", product }),
    setQty: (id, qty) => context.dispatch({ type: "setQty", id, qty }),
  };
}
```

```ts file=/storage.ts
import type { CartItem } from "./types";

const KEY = "minishop-cart";

export function loadCart(): CartItem[] {
  try {
    const saved: unknown = JSON.parse(localStorage.getItem(KEY) ?? "null");
    return Array.isArray(saved) ? (saved as CartItem[]) : [];
  } catch {
    return [];
  }
}

export function saveCart(items: CartItem[]): void {
  try {
    localStorage.setItem(KEY, JSON.stringify(items));
  } catch {
    // Storage is full or blocked (private mode): the cart still works, it just isn't saved.
  }
}
```

```tsx file=/CartContext.tsx
import {
  createContext,
  useContext,
  useEffect,
  useReducer,
  type Dispatch,
  type ReactNode,
} from "react";
import { cartReducer } from "./cartReducer";
import { cartCount, cartTotal } from "./cartUtils";
import { loadCart, saveCart } from "./storage";
import type { CartAction, CartItem, Product } from "./types";

interface CartContextValue {
  items: CartItem[];
  dispatch: Dispatch<CartAction>;
}

const CartContext = createContext<CartContextValue | null>(null);

export function CartProvider({ children }: { children: ReactNode }) {
  const [items, dispatch] = useReducer(cartReducer, undefined, loadCart);

  useEffect(() => {
    saveCart(items);
  }, [items]);

  return <CartContext value={{ items, dispatch }}>{children}</CartContext>;
}

export function useCart() {
  const context = useContext(CartContext);
  if (!context) throw new Error("useCart must be used inside <CartProvider>");
  const { items, dispatch } = context;
  return {
    items,
    count: cartCount(items),
    total: cartTotal(items),
    add: (product: Product) => dispatch({ type: "add", product }),
    remove: (id: number) => dispatch({ type: "remove", id }),
    setQty: (id: number, qty: number) => dispatch({ type: "setQty", id, qty }),
    clear: () => dispatch({ type: "clear" }),
  };
}
```

```ts code file=/storage.ts
  } catch {
    // Storage is full or blocked (private mode): the cart still works, it just isn't saved.
  }
}
```
