```js tests file=/cart.test.js
import { cartTotal, renderCart } from "./cartUtils";

const items = [
  { id: 1, name: "Coffee beans", price: 12, qty: 2 },
  { id: 2, name: "Green tea", price: 8, qty: 1 },
];

test("cartTotal adds price × qty for every item", () => {
  expect(cartTotal(items)).toBe(32);
});

test("an empty cart totals 0", () => {
  expect(cartTotal([])).toBe(0);
});

test("renderCart makes one <li> per item, with the line total", () => {
  document.body.innerHTML = `<ul>${renderCart(items)}</ul>`;
  const lines = [...document.querySelectorAll("li")].map((li) => li.textContent);
  expect(lines).toEqual(["Coffee beans × 2: $24.00", "Green tea × 1: $8.00"]);
});

test("renderCart returns one string, with no commas between the lines", () => {
  const html = renderCart(items);
  expect(typeof html).toBe("string");
  expect(html).not.toContain(",");
});

test("an empty cart says so", () => {
  document.body.innerHTML = `<ul>${renderCart([])}</ul>`;
  expect(document.querySelector("li").textContent).toBe("Your cart is empty.");
});
```

```js solution file=/cartUtils.js
export function formatPrice(amount) {
  return "$" + amount.toFixed(2);
}

export function cartTotal(items) {
  return items.reduce((sum, item) => sum + item.price * item.qty, 0);
}

export function renderCart(items) {
  if (items.length === 0) return "<li>Your cart is empty.</li>";
  return items
    .map((item) => `<li>${item.name} × ${item.qty}: ${formatPrice(item.price * item.qty)}</li>`)
    .join("");
}
```

<HowItWorks title="What innerHTML really does">

Setting `innerHTML` makes the browser **parse** your string as HTML, throw away everything that was inside the element, and build brand-new DOM nodes. It's fast to write, but it has two costs you'll meet soon:

- Anything inside is destroyed and recreated, including focus, text a user was typing and event listeners on those elements.
- The string is trusted as HTML. If a product name came from a user and contained `<img src=x onerror=…>`, it would run. (React escapes text for you, which is one more reason we'll switch to it.)

For a static cart that's fine. The trouble starts when the cart **changes**, in the next step.
