Reload the preview from step 5 and the cart is gone. Real shops remember it. The browser gives you `localStorage`, a small key-value store that survives reloads, and React gives you two tools to connect it:

- **Loading** happens once, when the cart state is created. `useReducer(reducer, initialArg, init)` calls `init` only on the first render: a *lazy initializer*.
- **Saving** is a side effect: it talks to something outside React. Side effects that follow a render belong in `useEffect`, which runs after React has updated the screen.

```js starter file=/cartUtils.js
export function formatPrice(amount) {
  return "$" + amount.toFixed(2);
}

export function cartTotal(items) {
  return items.reduce((sum, item) => sum + item.price * item.qty, 0);
}

export function cartCount(items) {
  return items.reduce((sum, item) => sum + item.qty, 0);
}

export function addToCart(items, product) {
  const existing = items.find((item) => item.id === product.id);
  if (existing) {
    return items.map((item) => (item.id === product.id ? { ...item, qty: item.qty + 1 } : item));
  }
  return [...items, { ...product, qty: 1 }];
}

export function renderCart(items) {
  if (items.length === 0) return "<li>Your cart is empty.</li>";
  return items
    .map((item) => `<li>${item.name} × ${item.qty}: ${formatPrice(item.price * item.qty)}</li>`)
    .join("");
}
```

```js starter file=/cartReducer.js
import { addToCart } from "./cartUtils";

export function cartReducer(items, action) {
  switch (action.type) {
    case "add":
      return addToCart(items, action.product);
    case "remove":
      return items.filter((item) => item.id !== action.id);
    case "setQty":
      if (action.qty < 1) return items.filter((item) => item.id !== action.id);
      return items.map((item) => (item.id === action.id ? { ...item, qty: action.qty } : item));
    case "clear":
      return [];
    default:
      throw new Error(`Unknown action: ${action.type}`);
  }
}
```

```jsx starter file=/Cart.js
                aria-label={`Increase ${item.name}`}

              </button>
              <button aria-label={`Remove ${item.name}`} onClick={() => onRemove(item.id)}>

              </button>
```

```js solution file=/storage.js
const KEY = "minishop-cart";

export function loadCart() {
  try {
    const saved = JSON.parse(localStorage.getItem(KEY));
    return Array.isArray(saved) ? saved : [];
  } catch {
    return [];
  }
}

export function saveCart(items) {
  try {
    localStorage.setItem(KEY, JSON.stringify(items));
  } catch {
    // Storage is full or blocked (private mode): the cart still works, it just isn't saved.
  }
}
```

```jsx solution file=/App.js
import { useEffect, useReducer } from "react";
import { products } from "./products";
import { cartReducer } from "./cartReducer";
import { loadCart, saveCart } from "./storage";
import ProductList from "./ProductList";
import Cart from "./Cart";

export default function App() {
  const [cart, dispatch] = useReducer(cartReducer, undefined, loadCart);

  useEffect(() => {
    saveCart(cart);
  }, [cart]);

  return (
    <main>
      <h1>MiniShop</h1>
      <ProductList products={products} onAdd={(product) => dispatch({ type: "add", product })} />
      <Cart
        items={cart}
        onChangeQty={(id, qty) => dispatch({ type: "setQty", id, qty })}
        onRemove={(id) => dispatch({ type: "remove", id })}
        onClear={() => dispatch({ type: "clear" })}
      />
    </main>
  );
}
```

<HowItWorks title="Why not save inside the reducer?">

It's tempting to write `localStorage.setItem(…)` at the end of each `case`. Don't: a reducer must be **pure**. React may run it twice in development, run it during a render that ends up thrown away, or run several queued actions in a row. Each of those would write to storage at the wrong time.

`useEffect` runs **after** React commits a render to the screen, with the cart that's actually showing. Its dependency array, `[cart]`, means "run again only when the cart is a different value". Thanks to immutable updates, every real change is a new array, so it always saves exactly what the user sees.

The lazy initializer has the same logic in reverse: reading storage once, at the start, instead of in every render.

</HowItWorks>
