Real shops don't ship their product list inside the JavaScript. It comes from a server, which takes time and sometimes fails. So the UI has three states, not one: **loading**, **error** and **success**.

`api.js` is a pretend server: `api.fetchProducts()` returns a promise that resolves after 600 ms, just like `fetch` would. You'll use it twice:

1. **By hand, with `useEffect`.** You'll see every moving part: state for the result, an effect to start the request, and a cleanup for when the component goes away.
2. **With TanStack Query**, the library most React teams use for server data. It handles the same states, plus caching, retries and refetching, in one line.

```js starter file=/cartUtils.js
export function formatPrice(amount) {
  return "$" + amount.toFixed(2);
}

export function cartTotal(items) {
  return items.reduce((sum, item) => sum + item.price * item.qty, 0);
}

export function cartCount(items) {
  return items.reduce((sum, item) => sum + item.qty, 0);
}

export function addToCart(items, product) {
  const existing = items.find((item) => item.id === product.id);
  if (existing) {
    return items.map((item) => (item.id === product.id ? { ...item, qty: item.qty + 1 } : item));
  }
  return [...items, { ...product, qty: 1 }];
}

export function renderCart(items) {
  if (items.length === 0) return "<li>Your cart is empty.</li>";
  return items
    .map((item) => `<li>${item.name} × ${item.qty}: ${formatPrice(item.price * item.qty)}</li>`)
    .join("");
}
```

```js starter file=/cartReducer.js
import { addToCart } from "./cartUtils";

export function cartReducer(items, action) {
  switch (action.type) {
    case "add":
      return addToCart(items, action.product);
    case "remove":
      return items.filter((item) => item.id !== action.id);
    case "setQty":
      if (action.qty < 1) return items.filter((item) => item.id !== action.id);
      return items.map((item) => (item.id === action.id ? { ...item, qty: action.qty } : item));
    case "clear":
      return [];
    default:
      throw new Error(`Unknown action: ${action.type}`);
  }
}
```

```js starter file=/storage.js
const KEY = "minishop-cart";

export function loadCart() {
  try {
    const saved = JSON.parse(localStorage.getItem(KEY));
    return Array.isArray(saved) ? saved : [];
  } catch {
    return [];
  }
}

export function saveCart(items) {
  try {
    localStorage.setItem(KEY, JSON.stringify(items));
  } catch {
    // Storage is full or blocked (private mode): the cart still works, it just isn't saved.
  }
}
```

```js solution file=/useProducts.js
import { useEffect, useState } from "react";
import { api } from "./api";

export function useProducts() {
  const [state, setState] = useState({ status: "loading", products: [] });

  useEffect(() => {
    let ignore = false;
    api.fetchProducts().then(
      (products) => {
        if (!ignore) setState({ status: "success", products });
      },
      () => {
        if (!ignore) setState({ status: "error", products: [] });
      },
    );
    return () => {
      ignore = true;
    };
  }, []);

  return state;
}
```

<HowItWorks title="What the hand-written version doesn't do">

Your `useProducts` works, but count what a real shop still needs:

- **Caching.** Every component that calls `useProducts()` starts its own request. Open the product page, go back, and the list loads again from scratch.
- **Retries and refetching.** A flaky network gets one chance. Coming back to the tab doesn't refresh stale prices.
- **Deduplication.** Two components asking at the same moment make two requests.

You could add all of it, and you'd have written a data-fetching library. TanStack Query is that library: it keeps server data in a shared cache keyed by `queryKey`, so every `useQuery({ queryKey: ["products"] })` in the app shares one request and one result.

</HowItWorks>

```jsx solution file=/App.js
import { useEffect, useReducer, useState } from "react";

import { cartReducer } from "./cartReducer";
import { loadCart, saveCart } from "./storage";
```

```jsx solution file=/ProductList.test.js
import { fireEvent, render, screen } from "@testing-library/react";
import "@testing-library/jest-dom";
import { QueryClient, QueryClientProvider } from "@tanstack/react-query";
import { afterEach, expect, test, vi } from "vitest";
import ProductList from "./ProductList";
import { api } from "./api";
import { products } from "./products";

function renderList(onAdd = vi.fn()) {
  const client = new QueryClient({ defaultOptions: { queries: { retry: false } } });
  render(
    <QueryClientProvider client={client}>
      <ProductList onAdd={onAdd} />
    </QueryClientProvider>,
  );
  return onAdd;
}

afterEach(() => vi.restoreAllMocks());

test("shows a loading message, then the products", async () => {
  vi.spyOn(api, "fetchProducts").mockResolvedValue(products);
  renderList();
  expect(screen.getByRole("status")).toHaveTextContent("Loading products…");
  expect(await screen.findByText(/Green tea/)).toBeInTheDocument();
});

test("Add reports the product that was clicked", async () => {
  vi.spyOn(api, "fetchProducts").mockResolvedValue(products);
  const onAdd = renderList();
  fireEvent.click(await screen.findByRole("button", { name: "Add Green tea" }));
  expect(onAdd).toHaveBeenCalledWith(products[1]);
});

test("a failed request shows an error, and Try again recovers", async () => {
  vi.spyOn(api, "fetchProducts")
    .mockRejectedValueOnce(new Error("down"))
    .mockResolvedValue(products);
  renderList();
  expect(await screen.findByRole("alert")).toHaveTextContent("Couldn't load the products.");
  fireEvent.click(screen.getByRole("button", { name: "Try again" }));
  expect(await screen.findByText(/Travel mug/)).toBeInTheDocument();
});
```
