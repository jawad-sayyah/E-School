`App` now holds three functions that each change the cart in their own way. As the shop grows (a clear button, a discount, a "buy again"), that logic spreads further. A **reducer** collects all of it in one place:

```js
function cartReducer(items, action) {
  // given the current cart and what happened, return the next cart
}
```

Components stop saying *how* the cart changes. They **dispatch** what happened (`{ type: "add", product }`), and the reducer decides. Because a reducer is a plain function with no React inside, you can test it with nothing but inputs and outputs.

```js starter file=/cartUtils.js
export function formatPrice(amount) {
  return "$" + amount.toFixed(2);
}

export function cartTotal(items) {
  return items.reduce((sum, item) => sum + item.price * item.qty, 0);
}

export function cartCount(items) {
  return items.reduce((sum, item) => sum + item.qty, 0);
}

export function addToCart(items, product) {
  const existing = items.find((item) => item.id === product.id);
  if (existing) {
    return items.map((item) => (item.id === product.id ? { ...item, qty: item.qty + 1 } : item));
  }
  return [...items, { ...product, qty: 1 }];
}

export function renderCart(items) {
  if (items.length === 0) return "<li>Your cart is empty.</li>";
  return items
    .map((item) => `<li>${item.name} × ${item.qty}: ${formatPrice(item.price * item.qty)}</li>`)
    .join("");
}
```

```jsx starter file=/App.js
import { useState } from "react";
import { products } from "./products";
import { addToCart } from "./cartUtils";
import ProductList from "./ProductList";
import Cart from "./Cart";

export default function App() {
  const [cart, setCart] = useState([]);

  function add(product) {
    setCart((items) => addToCart(items, product));
  }

  function changeQty(id, qty) {
    setCart((items) =>
      qty < 1
        ? items.filter((item) => item.id !== id)
        : items.map((item) => (item.id === id ? { ...item, qty } : item)),
    );
  }

  function remove(id) {
    setCart((items) => items.filter((item) => item.id !== id));
  }

  return (
    <main>
      <h1>MiniShop</h1>
      <ProductList products={products} onAdd={add} />
      <Cart items={cart} onChangeQty={changeQty} onRemove={remove} />
    </main>
  );
}
```

test("add: a new product joins with qty 1, an existing one gets +1", () => {
  const once = cartReducer([], { type: "add", product: tea });

  const items = [{ ...tea, qty: 1 }, { ...mug, qty: 2 }];
  expect(cartReducer(items, { type: "remove", id: 2 })).toEqual([{ ...mug, qty: 2 }]);

test("setQty changes one line, and below 1 removes it", () => {
  const items = [{ ...tea, qty: 1 }, { ...mug, qty: 2 }];
  expect(cartReducer(items, { type: "setQty", id: 3, qty: 5 })).toEqual([{ ...tea, qty: 1 }, { ...mug, qty: 5 }]);

test("the reducer never changes the old cart", () => {

```js solution file=/cartReducer.js
import { addToCart } from "./cartUtils";

export function cartReducer(items, action) {
  switch (action.type) {
    case "add":
      return addToCart(items, action.product);
    case "remove":
      return items.filter((item) => item.id !== action.id);
    case "setQty":
      if (action.qty < 1) return items.filter((item) => item.id !== action.id);
      return items.map((item) => (item.id === action.id ? { ...item, qty: action.qty } : item));
    case "clear":
      return [];
    default:
      throw new Error(`Unknown action: ${action.type}`);
  }
}
```

import { useReducer } from "react";
import { products } from "./products";
import { cartReducer } from "./cartReducer";
import ProductList from "./ProductList";

export default function App() {
  const [cart, dispatch] = useReducer(cartReducer, []);

      <ProductList products={products} onAdd={(product) => dispatch({ type: "add", product })} />

        onChangeQty={(id, qty) => dispatch({ type: "setQty", id, qty })}
        onRemove={(id) => dispatch({ type: "remove", id })}
        onClear={() => dispatch({ type: "clear" })}
      />
    </main>
  );
}

<HowItWorks title="dispatch, step by step">

`dispatch({ type: "setQty", id: 2, qty: 3 })` doesn't run your reducer straight away. React **queues** the action and schedules a re-render. During that render, `useReducer` takes the current cart and runs `cartReducer(cart, action)` for each queued action, in order, and the result becomes the new state.

That has two consequences:

- The reducer must be **pure**. React may call it more than once (StrictMode does it on purpose in development to catch impure reducers), so no `fetch`, no `localStorage`, no changing the old array.
- `dispatch` itself never changes, so you can pass it down as deep as you like without causing re-renders. You'll use that in step 8, when the cart moves into context.

</HowItWorks>

function cartReducer(items, action) {
  switch (action.type) {
    case "add": {
      const existing = items.find((item) => item.id === action.product.id);
      if (existing) {
        return items.map((item) =>
          item.id === action.product.id ? { ...item, qty: item.qty + 1 } : item,
        );
      }
      return [...items, { ...action.product, qty: 1 }];
    }
    case "remove":
      return items.filter((item) => item.id !== action.id);
    case "setQty":
      if (action.qty < 1) return items.filter((item) => item.id !== action.id);
      return items.map((item) => (item.id === action.id ? { ...item, qty: action.qty } : item));
    case "clear":
      return [];
    default:
      throw new Error(`Unknown action: ${action.type}`);
  }
}
