The last big feature: taking the order. A checkout form looks simple, but it's where users make the most mistakes, so it has to explain each one clearly. Two libraries do the heavy lifting:

- **Zod** describes what valid data looks like, once: `z.object({ name: …, email: z.email(…) })`. `z.infer` turns that schema into a TypeScript type, so the form's data is typed for free.
- **React Hook Form** connects the inputs (`register`), runs the validation on submit (through `zodResolver`), and gives you `errors` to show next to each field.

The page has three states: an **empty cart** (nothing to check out), the **form** with the order summary, and a **thank-you** message after the order is placed.

```ts starter file=/cartUtils.ts
import type { CartItem, Product } from "./types";

export function formatPrice(amount: number): string {
  return "$" + amount.toFixed(2);
}

export function cartTotal(items: CartItem[]): number {
  return items.reduce((sum, item) => sum + item.price * item.qty, 0);
}

export function cartCount(items: CartItem[]): number {
  return items.reduce((sum, item) => sum + item.qty, 0);
}

export function addToCart(items: CartItem[], product: Product): CartItem[] {
  const existing = items.find((item) => item.id === product.id);
  if (existing) {
    return items.map((item) => (item.id === product.id ? { ...item, qty: item.qty + 1 } : item));
  }
  return [...items, { ...product, qty: 1 }];
}
```

```ts starter file=/cartReducer.ts
import { addToCart } from "./cartUtils";
import type { CartAction, CartItem } from "./types";

export function cartReducer(items: CartItem[], action: CartAction): CartItem[] {
  switch (action.type) {
    case "add":
      return addToCart(items, action.product);
    case "remove":
      return items.filter((item) => item.id !== action.id);
    case "setQty":
      if (action.qty < 1) return items.filter((item) => item.id !== action.id);
      return items.map((item) => (item.id === action.id ? { ...item, qty: action.qty } : item));
    case "clear":
      return [];
    default: {
      const unhandled: never = action;
      throw new Error(`Unknown action: ${JSON.stringify(unhandled)}`);
    }
  }
}
```

```ts starter file=/storage.ts
import type { CartItem } from "./types";

const KEY = "minishop-cart";

export function loadCart(): CartItem[] {
  try {
    const saved: unknown = JSON.parse(localStorage.getItem(KEY) ?? "null");
    return Array.isArray(saved) ? (saved as CartItem[]) : [];
  } catch {
    return [];
  }
}

export function saveCart(items: CartItem[]): void {
  try {
    localStorage.setItem(KEY, JSON.stringify(items));
  } catch {
    // Storage is full or blocked (private mode): the cart still works, it just isn't saved.
  }
}
```

```tsx starter file=/CartContext.tsx
import {
  createContext,
  useContext,
  useEffect,
  useReducer,
  type Dispatch,
  type ReactNode,
} from "react";
import { cartReducer } from "./cartReducer";
import { cartCount, cartTotal } from "./cartUtils";
import { loadCart, saveCart } from "./storage";
import type { CartAction, CartItem, Product } from "./types";

interface CartContextValue {
  items: CartItem[];
  dispatch: Dispatch<CartAction>;
}

const CartContext = createContext<CartContextValue | null>(null);

export function CartProvider({ children }: { children: ReactNode }) {
  const [items, dispatch] = useReducer(cartReducer, undefined, loadCart);

  useEffect(() => {
    saveCart(items);
  }, [items]);

  return <CartContext value={{ items, dispatch }}>{children}</CartContext>;
}

export function useCart() {
  const context = useContext(CartContext);
  if (!context) throw new Error("useCart must be used inside <CartProvider>");
  const { items, dispatch } = context;
  return {
    items,
    count: cartCount(items),
    total: cartTotal(items),
    add: (product: Product) => dispatch({ type: "add", product }),
    remove: (id: number) => dispatch({ type: "remove", id }),
    setQty: (id: number, qty: number) => dispatch({ type: "setQty", id, qty }),
    clear: () => dispatch({ type: "clear" }),
  };
}
```

```tsx solution file=/pages/CheckoutPage.tsx
import { useState } from "react";
import { useForm } from "react-hook-form";
import { zodResolver } from "@hookform/resolvers/zod";
import { z } from "zod";
import { Link } from "react-router";
import { useCart } from "../CartContext";
import { formatPrice } from "../cartUtils";

const checkoutSchema = z.object({
  name: z.string().trim().min(2, "Enter your full name"),
  email: z.email("Enter a valid email"),
  address: z.string().trim().min(5, "Enter your street address"),
  zip: z.string().regex(/^\d{5}$/, "A ZIP code has 5 digits"),
});

type CheckoutForm = z.infer<typeof checkoutSchema>;

export default function CheckoutPage() {
  const { items, total, clear } = useCart();
  const [placedBy, setPlacedBy] = useState<string | null>(null);
  const {
    register,
    handleSubmit,
    formState: { errors },
  } = useForm<CheckoutForm>({ resolver: zodResolver(checkoutSchema) });

  if (placedBy) {
    return (
      <section>
        <h1>Thank you!</h1>
        <p role="status">Thanks, {placedBy}! Your order is on its way.</p>
        <Link to="/products">Keep shopping</Link>
      </section>
    );
  }

  if (items.length === 0) {
    return (
      <section>
        <h1>Checkout</h1>
        <p>Your cart is empty.</p>
        <Link to="/products">Find something you like</Link>
      </section>
    );
  }

  function placeOrder(data: CheckoutForm) {
    setPlacedBy(data.name);
    clear();
  }

  return (
    <section>
      <h1>Checkout</h1>
      <h2>Order summary</h2>
      <ul>
        {items.map((item) => (
          <li key={item.id}>
            {item.name} × {item.qty}: {formatPrice(item.price * item.qty)}
          </li>
        ))}
      </ul>
      <p>Total: {formatPrice(total)}</p>

      <form onSubmit={handleSubmit(placeOrder)} noValidate>
        <label htmlFor="name">Name</label>
        <input
          id="name"
          autoComplete="name"
          aria-invalid={errors.name ? "true" : "false"}
          {...register("name")}
        />
        {errors.name && <p role="alert">{errors.name.message}</p>}

        <label htmlFor="email">Email</label>
        <input
          id="email"
          type="email"
          autoComplete="email"
          aria-invalid={errors.email ? "true" : "false"}
          {...register("email")}
        />
        {errors.email && <p role="alert">{errors.email.message}</p>}

        <label htmlFor="address">Address</label>
        <input
          id="address"
          autoComplete="street-address"
          aria-invalid={errors.address ? "true" : "false"}
          {...register("address")}
        />
        {errors.address && <p role="alert">{errors.address.message}</p>}

        <label htmlFor="zip">ZIP code</label>
        <input
          id="zip"
          inputMode="numeric"
          autoComplete="postal-code"
          aria-invalid={errors.zip ? "true" : "false"}
          {...register("zip")}
        />
        {errors.zip && <p role="alert">{errors.zip.message}</p>}

        <button type="submit">Place order</button>
      </form>
    </section>
  );
}
```

<HowItWorks title="What happens when you press Place order">

1. The browser fires `submit` on the form. `handleSubmit` calls `event.preventDefault()`, so the page doesn't reload.
2. React Hook Form reads every registered input. It doesn't keep them in React state while you type (inputs are **uncontrolled**), which is why typing doesn't re-render the whole page.
3. `zodResolver` runs `checkoutSchema.safeParse(values)`. Zod checks every field and collects **all** the problems, not just the first.
4. If there are problems, they become `errors` (one per field, with your messages), the component re-renders to show them, and the first invalid field gets focus. `placeOrder` is not called.
5. If everything passes, `placeOrder` receives the **parsed** data. Thanks to `.trim()`, `"  Ada "` arrives as `"Ada"`, and its type is exactly `CheckoutForm`.

A real shop would now `await` a request to its server, and the server would validate the same data again: anyone can bypass your form and send whatever they like.

</HowItWorks>

```tsx code file=/CartContext.tsx
import {
  createContext,
  useContext,
  useEffect,
  useReducer,
  type Dispatch,
  type ReactNode,
} from "react";
import { cartReducer } from "./cartReducer";
import { cartCount, cartTotal } from "./cartUtils";
import { loadCart, saveCart } from "./storage";
import type { CartAction, CartItem, Product } from "./types";

interface CartContextValue {
  items: CartItem[];
  dispatch: Dispatch<CartAction>;
}

const CartContext = createContext<CartContextValue | null>(null);

export function CartProvider({ children }: { children: ReactNode }) {
  const [items, dispatch] = useReducer(cartReducer, undefined, loadCart);

  useEffect(() => {
    saveCart(items);
  }, [items]);

  return <CartContext value={{ items, dispatch }}>{children}</CartContext>;
}

export function useCart() {
  const context = useContext(CartContext);
  if (!context) throw new Error("useCart must be used inside <CartProvider>");
  const { items, dispatch } = context;
  return {
    items,
    count: cartCount(items),
    total: cartTotal(items),
    add: (product: Product) => dispatch({ type: "add", product }),
    remove: (id: number) => dispatch({ type: "remove", id }),
    setQty: (id: number, qty: number) => dispatch({ type: "setQty", id, qty }),
    clear: () => dispatch({ type: "clear" }),
  };
}
```

```ts code file=/cartReducer.ts
import { addToCart } from "./cartUtils";
import type { CartAction, CartItem } from "./types";

export function cartReducer(items: CartItem[], action: CartAction): CartItem[] {
  switch (action.type) {
    case "add":
      return addToCart(items, action.product);
    case "remove":
      return items.filter((item) => item.id !== action.id);
    case "setQty":
      if (action.qty < 1) return items.filter((item) => item.id !== action.id);
      return items.map((item) => (item.id === action.id ? { ...item, qty: action.qty } : item));
    case "clear":
      return [];
    default: {
      const unhandled: never = action;
      throw new Error(`Unknown action: ${JSON.stringify(unhandled)}`);
    }
  }
}
```

```ts code file=/cartUtils.ts
import type { CartItem, Product } from "./types";

export function formatPrice(amount: number): string {
  return "$" + amount.toFixed(2);
}

export function cartTotal(items: CartItem[]): number {
  return items.reduce((sum, item) => sum + item.price * item.qty, 0);
}

export function cartCount(items: CartItem[]): number {
  return items.reduce((sum, item) => sum + item.qty, 0);
}

export function addToCart(items: CartItem[], product: Product): CartItem[] {
  const existing = items.find((item) => item.id === product.id);
  if (existing) {
    return items.map((item) => (item.id === product.id ? { ...item, qty: item.qty + 1 } : item));
  }
  return [...items, { ...product, qty: 1 }];
}
```

```ts code file=/storage.ts
import type { CartItem } from "./types";

const KEY = "minishop-cart";

export function loadCart(): CartItem[] {
  try {
    const saved: unknown = JSON.parse(localStorage.getItem(KEY) ?? "null");
    return Array.isArray(saved) ? (saved as CartItem[]) : [];
  } catch {
    return [];
  }
}

export function saveCart(items: CartItem[]): void {
  try {
    localStorage.setItem(KEY, JSON.stringify(items));
  } catch {
    // Storage is full or blocked (private mode): the cart still works, it just isn't saved.
  }
}
```
