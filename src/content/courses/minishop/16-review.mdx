<SpotTheBug
  id="bug-review-minishop"
  bug={[4, 5]}
  prompt="After placing an order, the page says 'Your cart is empty.' instead of thanking the customer. Click the lines that cause it."
  explain="The empty-cart check runs first, and `clear()` has just emptied the cart, so the thank-you branch is never reached. Check `placedBy` before `items.length` (step 11)."
>

```jsx
function CheckoutPage() {
  const { items, clear } = useCart();
  const [placedBy, setPlacedBy] = useState(null);
  if (items.length === 0) {
    return <p>Your cart is empty.</p>;
  }
  if (placedBy) {
    return <p>Thanks, {placedBy}! Your order is on its way.</p>;
  }
  return <CheckoutForm onOrder={(name) => { setPlacedBy(name); clear(); }} />;
}
```
