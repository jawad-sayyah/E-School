<Exercise

Write `searchProducts(products, query)` for the new search box. It ignores case and spaces around the query, matches any part of the name, returns the results from cheapest to most expensive, returns every product (sorted) for an empty query, and never changes the `products` array it was given.

```js starter
function searchProducts(products, query) {
  return products.filter((p) => p.name.includes(query));
}

const products = [
  { id: 3, name: "Travel mug", price: 15 },
  { id: 1, name: "Coffee beans", price: 12 },
  { id: 2, name: "Green tea", price: 8 },
];
console.log(searchProducts(products, " TEA "));
```

```js solution
function searchProducts(products, query) {
  const q = query.trim().toLowerCase();
  const found = q ? products.filter((p) => p.name.toLowerCase().includes(q)) : products;
  return [...found].sort((a, b) => a.price - b.price);
}

const products = [
  { id: 3, name: "Travel mug", price: 15 },
  { id: 1, name: "Coffee beans", price: 12 },
  { id: 2, name: "Green tea", price: 8 },
];
console.log(searchProducts(products, " TEA "));
```

<TSExercise
  id="boss-coupons"
  title="Typed coupons"
  hints={[
    "`type Coupon = { kind: \"percent\"; percent: number } | { kind: \"fixed\"; amount: number } | { kind: \"freeShipping\" };`",
    "Handle `null` first, then `switch (coupon.kind)`: percent is `subtotal * (1 - coupon.percent / 100) + shipping`, fixed is `Math.max(0, subtotal - coupon.amount) + shipping`, free shipping is just `subtotal`.",
    "Finish with ``default: { const unhandled: never = coupon; throw new Error(`Unknown coupon: ${JSON.stringify(unhandled)}`); }``.",
  ]}
>

Marketing wants three kinds of coupon: a **percent** off the products, a **fixed** amount off the products (never below zero), and **free shipping**. Model `Coupon` as a discriminated union, and write `orderTotal(subtotal, shipping, coupon)` (the coupon can be `null`). The `switch` must be exhaustive.

```ts starter
type Coupon = any;

function orderTotal(subtotal: number, shipping: number, coupon: Coupon | null): number {
  return subtotal + shipping;
}
```

```ts types
const total: number = orderTotal(40, 5, { kind: "percent", percent: 10 });
orderTotal(40, 5, { kind: "fixed", amount: 10 });
orderTotal(40, 5, { kind: "freeShipping" });
orderTotal(40, 5, null);
// @ts-expect-error there is no "bogo" coupon
orderTotal(40, 5, { kind: "bogo" });
// @ts-expect-error a percent coupon needs its percent
orderTotal(40, 5, { kind: "percent" });
// @ts-expect-error amounts are numbers
orderTotal(40, 5, { kind: "fixed", amount: "10" });
```

```js tests
test("no coupon: products plus shipping", () => {
  expect(orderTotal(40, 5, null)).toBe(45);
});
test("percent off the products only", () => {
  expect(orderTotal(40, 5, { kind: "percent", percent: 25 })).toBe(35);
});
test("fixed amount off, never below zero", () => {
  expect(orderTotal(40, 5, { kind: "fixed", amount: 10 })).toBe(35);
  expect(orderTotal(8, 5, { kind: "fixed", amount: 10 })).toBe(5);
});
test("free shipping", () => {
  expect(orderTotal(40, 5, { kind: "freeShipping" })).toBe(40);
});
```

```ts solution
type Coupon =
  | { kind: "percent"; percent: number }
  | { kind: "fixed"; amount: number }
  | { kind: "freeShipping" };

function orderTotal(subtotal: number, shipping: number, coupon: Coupon | null): number {
  if (coupon === null) return subtotal + shipping;
  switch (coupon.kind) {
    case "percent":
      return subtotal * (1 - coupon.percent / 100) + shipping;
    case "fixed":
      return Math.max(0, subtotal - coupon.amount) + shipping;
    case "freeShipping":
      return subtotal;
    default: {
      const unhandled: never = coupon;
      throw new Error(`Unknown coupon: ${JSON.stringify(unhandled)}`);
    }
  }
}
```

```jsx starter file=/App.js
import { useWishlist } from "./useWishlist";
import WishlistButton from "./WishlistButton";

const products = [
  { id: 1, name: "Coffee beans", price: 12 },
  { id: 2, name: "Green tea", price: 8 },
  { id: 3, name: "Travel mug", price: 15 },
];

export default function App() {
  const { ids, has, toggle } = useWishlist();
  return (
    <main>
      <h1>MiniShop</h1>
      <p>Saved for later: {ids.length}</p>
      <ul>
        {products.map((product) => (
          <li key={product.id}>
            {product.name}{" "}
            <WishlistButton
              name={product.name}
              saved={has(product.id)}
              onToggle={() => toggle(product.id)}
            />
          </li>
        ))}
      </ul>
    </main>
  );
}
```

beforeEach(() => localStorage.clear());

```jsx solution file=/useWishlist.js
import { useEffect, useState } from "react";

const KEY = "minishop-wishlist";

function loadIds() {
  try {
    const saved = JSON.parse(localStorage.getItem(KEY));
    return Array.isArray(saved) ? saved : [];
  } catch {
    return [];
  }
}

export function useWishlist() {
  const [ids, setIds] = useState(loadIds);

  useEffect(() => {
    localStorage.setItem(KEY, JSON.stringify(ids));
  }, [ids]);

  return {
    ids,
    has: (id) => ids.includes(id),
    toggle: (id) =>
      setIds((current) =>
        current.includes(id) ? current.filter((x) => x !== id) : [...current, id],
      ),
  };
}
```

```jsx solution file=/WishlistButton.js
export default function WishlistButton({ name, saved, onToggle }) {
  return (
    <button aria-pressed={saved} aria-label={`Save ${name}`} onClick={onToggle}>
      {saved ? "♥ Saved" : "♡ Save"}
    </button>
  );
}
```
