  explain="Line 1: an action used with `useActionState` receives `(previousState, formData)`, so `formData` here is really the previous state. Line 9: `useFormStatus` is called in the component that renders the form, so it never sees it pending; move the button into a child component."

```jsx
async function joinAction(formData) {
  const email = formData.get("email");
  if (!email) return { error: "Email required" };
  await join(email);
  return { joined: email };
}

function JoinForm() {
  const { pending } = useFormStatus();
  const [state, formAction] = useActionState(joinAction, {});
  return (
    <form action={formAction}>
      <input name="email" />
      <button disabled={pending}>Join</button>
      {state.error && <p>{state.error}</p>}
    </form>
  );
}
```
