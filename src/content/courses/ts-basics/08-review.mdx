<TSExercise
  id="typed-cart"
  title="A fully typed cart"
  hints={[
    "`type CartLine = { id: number; name: string; price: number; qty: number };`",
    "`function addLine(cart: readonly CartLine[], line: CartLine): CartLine[]` returns `[...cart, line]`; `removeLine` uses `filter`.",
    "`findLine(cart: readonly CartLine[], id: number): CartLine | undefined` returns `cart.find((l) => l.id === id)`.",
  ]}
>

Type the cart helpers: define `CartLine` (`id`, `name`, `price`, `qty`), then give `addLine`, `removeLine` and `findLine` full signatures. The cart parameter must be **readonly** (they return new arrays), and `findLine` must admit it may find nothing.

```ts starter
type CartLine = any;

function addLine(cart, line) {
  return [...cart, line];
}

function removeLine(cart, id) {
  return cart.filter((l) => l.id !== id);
}

function findLine(cart, id) {
  return cart.find((l) => l.id === id);
}
```

```ts types
const cart: CartLine[] = [{ id: 1, name: "Tea", price: 2, qty: 1 }];
const next: CartLine[] = addLine(cart, { id: 2, name: "Bagel", price: 3, qty: 1 });
const fewer: CartLine[] = removeLine(next, 1);
const found = findLine(cart, 1);
// a found line may be undefined, so this must be checked first
const name: string = found?.name ?? "none";
// @ts-expect-error findLine may return undefined
const unsafe: string = findLine(cart, 1).name;
// @ts-expect-error a cart line needs a quantity
addLine(cart, { id: 3, name: "Cake", price: 5 });
// @ts-expect-error ids are numbers
removeLine(cart, "1");
```

```js tests
const tea = { id: 1, name: "Tea", price: 2, qty: 1 };
const bagel = { id: 2, name: "Bagel", price: 3, qty: 1 };

test("addLine returns a new array with the line", () => {
  const cart = [tea];
  const next = addLine(cart, bagel);
  expect(next).toEqual([tea, bagel]);
  expect(cart).toEqual([tea]);
});

test("removeLine and findLine", () => {
  expect(removeLine([tea, bagel], 1)).toEqual([bagel]);
  expect(findLine([tea, bagel], 2)).toEqual(bagel);
  expect(findLine([tea], 9)).toBe(undefined);
});
```

```ts solution
type CartLine = { id: number; name: string; price: number; qty: number };

function addLine(cart: readonly CartLine[], line: CartLine): CartLine[] {
  return [...cart, line];
}

function removeLine(cart: readonly CartLine[], id: number): CartLine[] {
  return cart.filter((l) => l.id !== id);
}

function findLine(cart: readonly CartLine[], id: number): CartLine | undefined {
  return cart.find((l) => l.id === id);
}
```
