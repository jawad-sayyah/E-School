```jsx
function Clock() {
  const [seconds, setSeconds] = useState(0);
  useEffect(() => {
    const id = setInterval(() => {
      setSeconds(seconds + 1);
    }, 1000);
    return () => clearInterval(id);
  }, []);
  return <p>{seconds}s</p>;
}
```

Build a reusable accordion item:

- write a `useToggle(initial = false)` hook in `useToggle.js` that returns `[on, toggle]`,
- in `FaqItem`, each question is a button with `aria-expanded` (true/false) and `aria-controls` pointing at its answer panel's id (use `useId`),
- clicking a question shows or hides only **its own** answer.

```jsx starter file=/useToggle.js
export function useToggle(initial = false) {
  return [initial, () => {}];
}
```

```jsx starter file=/FaqItem.js
import { useToggle } from "./useToggle";

export default function FaqItem({ question, answer }) {
  const [open, toggle] = useToggle();
  return (
    <div className="faq">
      <button>{question}</button>
      {open && <div>{answer}</div>}
    </div>
  );
}
```

```jsx starter file=/App.js
import FaqItem from "./FaqItem";

export default function App() {
  return (
    <>
      <FaqItem question="Do you have oat milk?" answer="Yes, at no extra cost." />
      <FaqItem question="Can I order ahead?" answer="Yes, in the app, up to a day before." />
    </>
  );
}
```

```jsx tests file=/App.test.js
import { fireEvent, render, screen } from "@testing-library/react";
import "@testing-library/jest-dom";
import App from "./App";

const button = (name) => screen.getByRole("button", { name });

test("answers start hidden and collapsed", () => {
  render(<App />);
  expect(button("Do you have oat milk?")).toHaveAttribute("aria-expanded", "false");
  expect(screen.queryByText("Yes, at no extra cost.")).not.toBeInTheDocument();
});

test("clicking a question opens only its answer", () => {
  render(<App />);
  fireEvent.click(button("Do you have oat milk?"));
  expect(button("Do you have oat milk?")).toHaveAttribute("aria-expanded", "true");
  expect(screen.getByText("Yes, at no extra cost.")).toBeInTheDocument();
  expect(screen.queryByText("Yes, in the app, up to a day before.")).not.toBeInTheDocument();
});

test("aria-controls points at the answer panel", () => {
  render(<App />);
  fireEvent.click(button("Can I order ahead?"));
  const id = button("Can I order ahead?").getAttribute("aria-controls");
  expect(id).toBeTruthy();
  expect(document.getElementById(id)).toHaveTextContent("Yes, in the app, up to a day before.");
});

test("clicking again closes it", () => {
  render(<App />);
  fireEvent.click(button("Do you have oat milk?"));
  fireEvent.click(button("Do you have oat milk?"));
  expect(screen.queryByText("Yes, at no extra cost.")).not.toBeInTheDocument();
});
```

```jsx solution file=/useToggle.js
import { useState } from "react";

export function useToggle(initial = false) {
  const [on, setOn] = useState(initial);
  return [on, () => setOn((v) => !v)];
}
```

```jsx solution file=/FaqItem.js
import { useId } from "react";
import { useToggle } from "./useToggle";

export default function FaqItem({ question, answer }) {
  const [open, toggle] = useToggle();
  const panelId = useId();
  return (
    <div className="faq">
      <button aria-expanded={open} aria-controls={panelId} onClick={toggle}>
        {question}
      </button>
      {open && <div id={panelId}>{answer}</div>}
    </div>
  );
}
```
