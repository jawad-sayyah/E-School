function SalesPage() {
  const Chart = lazy(() => import("./Chart"));
  const tableRef = useRef(null);

  return (
    <>
      <p>Table width: {tableRef.current.offsetWidth}px</p>
      <table ref={tableRef}>…</table>
      <Suspense fallback={<p>Loading chart…</p>}>
        <Chart />
      </Suspense>
    </>
  );
}
