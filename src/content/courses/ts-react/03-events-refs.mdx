Events and refs are where people reach for `any` most. You don't need to: React's types describe every event and element.

<Concept name="Inline handlers infer everything" id="inline">

When you write the handler inline, TypeScript knows the event type from the prop:

```tsx file=Search.tsx
<input onChange={(e) => setQuery(e.target.value)} />   // e: ChangeEvent<HTMLInputElement>
<button onClick={(e) => e.stopPropagation()} />          // e: MouseEvent<HTMLButtonElement>
```

Hover `e` in the editor to see it.

</Concept>

<Concept name="Separate handlers need the event type" id="separate">

A handler written as its own function has no context, so type its parameter. The pattern is `React.SomethingEvent<TheElement>`:

<TSPlayground file="CouponForm.tsx" errors={[13]} checkOnly>

```tsx
import { useState, type ChangeEvent, type FormEvent } from "react";

function CouponForm() {
  const [code, setCode] = useState("");

  function handleChange(e: ChangeEvent<HTMLInputElement>) {
    setCode(e.target.value.toUpperCase());
  }

  function handleSubmit(e: FormEvent<HTMLFormElement>) {
    e.preventDefault();
    console.log("apply", code);
    setCode(e.target.value);       // error: a form has no value
  }

  return (
    <form onSubmit={handleSubmit}>
      <input value={code} onChange={handleChange} />
    </form>
  );
}
```

</TSPlayground>

| Event | Type |
| --- | --- |
| typing in an input | `ChangeEvent<HTMLInputElement>` (`HTMLSelectElement`, `HTMLTextAreaElement`…) |
| submitting a form | `FormEvent<HTMLFormElement>` |
| clicking | `MouseEvent<HTMLButtonElement>` |
| a key | `KeyboardEvent<HTMLInputElement>` |
| focus / blur | `FocusEvent<HTMLInputElement>` |

These are React's types (import them from `"react"`), not the browser's global `MouseEvent`.

</Concept>

<Concept name="Refs" id="refs">

A DOM ref starts as `null` (the element doesn't exist during the first render), so its type is `HTMLInputElement | null`:

<TSPlayground file="SearchBox.tsx" errors={[7]} checkOnly>

```tsx
import { useRef } from "react";

function SearchBox() {
  const inputRef = useRef<HTMLInputElement>(null);   // RefObject<HTMLInputElement | null>

  function focus() {
    inputRef.current.focus();                          // error: current may be null
    inputRef.current?.focus();                         // fine
  }

  const timer = useRef<number | undefined>(undefined); // a value ref, not a DOM ref
  timer.current = window.setTimeout(focus, 1000);

  return <input ref={inputRef} />;
}
```

</TSPlayground>

Pass the **element** type to `useRef` and `null` as the initial value. In React 19, a component that accepts `ref` as a prop types it as `ref?: Ref<HTMLInputElement>`.

</Concept>

<HowItWorks title="Where do HTMLInputElement and friends come from?">

They're the browser's DOM types from TypeScript's `lib.dom.d.ts` (that's why `"DOM"` is in `lib`). Each element type lists its properties: `HTMLInputElement` has `value` and `checked`, `HTMLFormElement` has `elements` and `reset()`, `HTMLButtonElement` has `disabled`. `e.target` in a `ChangeEvent<HTMLInputElement>` is typed as that element, which is how `e.target.value` gets autocompleted and checked.

</HowItWorks>

  title="Type the quantity form"

Remove every implicit `any`: type the three handlers and the ref. The size must stay a union, so convert carefully from the select's string value (there's a helper).

```tsx starter
import { useRef, useState, type ChangeEvent, type FormEvent } from "react";

type Size = "small" | "large";
const isSize = (v: string): v is Size => v === "small" || v === "large";

export function QtyForm({ onOrder }: { onOrder: (qty: number, size: Size) => void }) {
  const [qty, setQty] = useState(1);
  const [size, setSize] = useState<Size>("small");
  const qtyRef = useRef(null);

  function handleQty(e) {
    setQty(Number(e.target.value));
  }
  function handleSize(e) {
    if (isSize(e.target.value)) setSize(e.target.value);
  }
  function handleSubmit(e) {
    e.preventDefault();
    if (qty < 1) return qtyRef.current.focus();
    onOrder(qty, size);
  }

  return (
    <form onSubmit={handleSubmit}>
      <input ref={qtyRef} type="number" value={qty} onChange={handleQty} />
      <select value={size} onChange={handleSize}>
        <option value="small">Small</option>
        <option value="large">Large</option>
      </select>
      <button>Order</button>
    </form>
  );
}
```

```tsx solution
import { useRef, useState, type ChangeEvent, type FormEvent } from "react";

type Size = "small" | "large";
const isSize = (v: string): v is Size => v === "small" || v === "large";

  const [size, setSize] = useState<Size>("small");
  const qtyRef = useRef<HTMLInputElement>(null);

  function handleQty(e: ChangeEvent<HTMLInputElement>) {
    setQty(Number(e.target.value));
  }
  function handleSize(e: ChangeEvent<HTMLSelectElement>) {
    if (isSize(e.target.value)) setSize(e.target.value);
  }
  function handleSubmit(e: FormEvent<HTMLFormElement>) {
    e.preventDefault();
    if (qty < 1) return qtyRef.current?.focus();
    onOrder(qty, size);
  }

  return (
    <form onSubmit={handleSubmit}>
      <input ref={qtyRef} type="number" value={qty} onChange={handleQty} />
      <select value={size} onChange={handleSize}>
        <option value="small">Small</option>
        <option value="large">Large</option>
      </select>
      <button>Order</button>
    </form>
  );
}
```
