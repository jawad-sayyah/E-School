<SpotTheBug
  id="bug-review-ts-react"
  bug={[8, 11]}
  prompt="TypeScript reports two errors. Click either line. (Typing state, Typing context)"
  explain="Line 8: `items` came from `useState([])`, which is `never[]`, so adding a product is an error; write `useState<Product[]>([])`. Line 11: `ctx` may be null; use a guarding hook instead of `useContext` directly."
>

```tsx
import { createContext, useContext, useState } from "react";

type Product = { id: number; name: string };
const CartContext = createContext<{ count: number } | null>(null);

function Shop() {
  const [items, setItems] = useState([]);
  const add = (p: Product) => setItems([...items, p]);

  const ctx = useContext(CartContext);
  return <p>{ctx.count} items · {items.length} shown</p>;
}
```
