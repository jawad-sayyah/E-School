<ReactExercise
  id="filter-products"
  title="Search the menu"
  hints={[
    "Add `const [query, setQuery] = useState(\"\");` and control the input with `value` and `onChange`.",
    "Derive the list: `const shown = products.filter((p) => p.name.toLowerCase().includes(query.trim().toLowerCase()));`",
    "Render `shown.map((p) => <li key={p.id}>{p.name}</li>)`, and the message when `shown.length === 0`.",
  ]}
>

Make the search box filter the menu as the user types:

- only products whose name contains the search text are listed (ignore upper/lower case),
- when nothing matches, show **No products match.** instead of the list,
- use the product ids as keys, and don't store the filtered list in state: derive it.

```jsx starter file=/App.js
const products = [
  { id: 1, name: "Coffee" },
  { id: 2, name: "Iced Coffee" },
  { id: 3, name: "Tea" },
  { id: 4, name: "Bagel" },
];

export default function App() {
  return (
    <section>
      <label>
        Search <input type="search" />
      </label>
      <ul>
        {products.map((p, i) => (
          <li key={i}>{p.name}</li>
        ))}
      </ul>
    </section>
  );
}
```

```jsx tests file=/App.test.js
import { fireEvent, render, screen } from "@testing-library/react";
import "@testing-library/jest-dom";
import App from "./App";

const names = () => screen.queryAllByRole("listitem").map((li) => li.textContent);
const search = (text) =>
  fireEvent.change(screen.getByLabelText("Search"), { target: { value: text } });

test("shows every product at first", () => {
  render(<App />);
  expect(names()).toEqual(["Coffee", "Iced Coffee", "Tea", "Bagel"]);
});

test("filters as you type, ignoring case", () => {
  render(<App />);
  search("COFFEE");
  expect(names()).toEqual(["Coffee", "Iced Coffee"]);
  expect(screen.getByLabelText("Search")).toHaveValue("COFFEE");
});

test("shows a message when nothing matches", () => {
  render(<App />);
  search("pizza");
  expect(names()).toEqual([]);
  expect(screen.getByText("No products match.")).toBeInTheDocument();
});

test("clearing the search shows everything again", () => {
  render(<App />);
  search("tea");
  search("");
  expect(names()).toHaveLength(4);
});
```

```jsx solution file=/App.js
import { useState } from "react";

const products = [
  { id: 1, name: "Coffee" },
  { id: 2, name: "Iced Coffee" },
  { id: 3, name: "Tea" },
  { id: 4, name: "Bagel" },
];

export default function App() {
  const [query, setQuery] = useState("");
  const q = query.trim().toLowerCase();
  const shown = products.filter((p) => p.name.toLowerCase().includes(q));

  return (
    <section>
      <label>
        Search{" "}
        <input type="search" value={query} onChange={(e) => setQuery(e.target.value)} />
      </label>
      {shown.length === 0 ? (
        <p>No products match.</p>
      ) : (
        <ul>
          {shown.map((p) => (
            <li key={p.id}>{p.name}</li>
          ))}
        </ul>
      )}
    </section>
  );
}
```
