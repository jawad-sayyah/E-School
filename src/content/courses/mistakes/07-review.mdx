  explain="Line 4: an effect that reads `category` but doesn't list it. Line 7: derived state (`count`) synced by an effect. Line 13: reading `selected` right after setting it."

```jsx
function Shop({ category }) {
  const [items, setItems] = useState([]);
  const [count, setCount] = useState(0);
  useEffect(() => {
    load(category).then(setItems);
  }, []);
  useEffect(() => setCount(items.length), [items]);

  const [selected, setSelected] = useState(null);

  function choose(item) {
    setSelected(item);
    track("selected", selected);
  }
  return <List items={items} count={count} onChoose={choose} />;
}
```
