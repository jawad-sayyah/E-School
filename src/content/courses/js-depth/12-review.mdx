<CodeReview
  id="review-unique-ids"
  answer={2}
  prompt="Which `uniqueIds(orders)` would you approve? (Map & Set)"
  reasons={[
    "Correct, but `includes` scans the whole result array for every order, so it gets slow with many orders.",
    "`filter` with `indexOf` works too, but it's just as slow and harder to read.",
    "A Set drops duplicates in one fast pass, and the spread turns it back into an array.",
  ]}
>

```js
function uniqueIds(orders) {
  const ids = [];
  for (const o of orders) {
    if (!ids.includes(o.id)) ids.push(o.id);
  }
  return ids;
}
```

```js
function uniqueIds(orders) {
  const ids = orders.map((o) => o.id);
  return ids.filter((id, i) => ids.indexOf(id) === i);
}
```

```js
function uniqueIds(orders) {
  return [...new Set(orders.map((o) => o.id))];
}
```

</CodeReview>

<Exercise
  id="review-price-tags"
  title="Price tags from a menu"
  hints={[
    "Find every `name: price` pair with `matchAll` and a regex like `/(\\w+): (\\d+(?:\\.\\d+)?)/g`.",
    "Build a Map from name to the formatted price.",
    "Format prices with `new Intl.NumberFormat(\"en-US\", { style: \"currency\", currency: \"USD\" })`.",
  ]}
>

Write `priceTags(menu)`. `menu` is text like `"Tea: 2, Cake: 4.5"`. Return a **Map** from each item name to its price formatted as US dollars, like `"$2.00"`.

```js starter
function priceTags(menu) {
  return new Map();
}

console.log(priceTags("Tea: 2, Cake: 4.5"));
```

```js tests
test("returns a Map", () => {
  expect(priceTags("") instanceof Map).toBe(true);
});

test("formats each price", () => {
  const tags = priceTags("Tea: 2, Cake: 4.5");
  expect(tags.get("Tea")).toBe("$2.00");
  expect(tags.get("Cake")).toBe("$4.50");
});

test("keeps every item", () => {
  expect(priceTags("A: 1, B: 2, C: 3").size).toBe(3);
});
```

```js solution
function priceTags(menu) {
  const usd = new Intl.NumberFormat("en-US", { style: "currency", currency: "USD" });
  const tags = new Map();
  for (const m of menu.matchAll(/(\w+): (\d+(?:\.\d+)?)/g)) {
    tags.set(m[1], usd.format(Number(m[2])));
  }
  return tags;
}

console.log(priceTags("Tea: 2, Cake: 4.5"));
```
