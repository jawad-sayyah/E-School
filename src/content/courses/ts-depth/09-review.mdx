<TSExercise

  title="A typed request helper"
  hints={[

    "Inside: `JSON.parse` in a try/catch; if `isT(value)` return `{ ok: true, data: value }`, otherwise `{ ok: false, error: \"Unexpected shape\" }`.",

>

Write a generic `parse(json, isT)` that parses a JSON string and validates it with a type guard. It returns a `Result<T>`: `{ ok: true, data }` with the **typed** data, or `{ ok: false, error }` for invalid JSON (**"Invalid JSON"**) or the wrong shape (**"Unexpected shape"**). `isProduct` is already written.

```ts starter
type Product = { name: string; price: number };

function isProduct(v: unknown): v is Product {
  return typeof v === "object" && v !== null && typeof (v as Product).name === "string" && typeof (v as Product).price === "number";
}

type Result<T> = any;

function parse(json: string, isT: (v: unknown) => boolean): any {
  return JSON.parse(json);
}
```

```ts types
const r = parse('{"name":"Tea","price":2}', isProduct);
if (r.ok) {
  // data is a Product here
  const price: number = r.data.price;
} else {
  const message: string = r.error;
}
// @ts-expect-error data only exists when ok is true
r.data;
```

```js tests
test("returns typed data for valid JSON of the right shape", () => {
  expect(parse('{"name":"Tea","price":2}', isProduct)).toEqual({ ok: true, data: { name: "Tea", price: 2 } });
});

test("reports invalid JSON", () => {
  expect(parse("{oops", isProduct)).toEqual({ ok: false, error: "Invalid JSON" });
});

test("reports the wrong shape", () => {
  expect(parse('{"title":"Tea"}', isProduct)).toEqual({ ok: false, error: "Unexpected shape" });
});
```

```ts solution
type Product = { name: string; price: number };

function isProduct(v: unknown): v is Product {
  return typeof v === "object" && v !== null && typeof (v as Product).name === "string" && typeof (v as Product).price === "number";
}

type Result<T> = { ok: true; data: T } | { ok: false; error: string };

function parse<T>(json: string, isT: (v: unknown) => v is T): Result<T> {
  let value: unknown;
  try {
    value = JSON.parse(json);
  } catch {
    return { ok: false, error: "Invalid JSON" };
  }
  return isT(value) ? { ok: true, data: value } : { ok: false, error: "Unexpected shape" };
}
```
