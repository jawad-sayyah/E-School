```jsx
function Products({ category }) {
  const add = useCartStore((s) => s.add);
  const { data = [] } = useQuery({
    queryKey: ["products"],
    queryFn: () => fetchProducts(category),
  });
  return data.map((p) => (
    <p key={p.id}>
      {p.name} <button onClick={() => add(p)}><PlusIcon /></button>
    </p>
  ));
}
```
