Build the cart with `useReducer`: **Add** buttons for each product (adding the same product twice increases its quantity), a cart list with a **Remove \{name\}** button per line, a badge with the item count (sum of quantities) and the **Total: $…** with two decimals.

```jsx starter file=/App.js
const PRODUCTS = [
  { id: 1, name: "Coffee", price: 4 },
  { id: 2, name: "Tea", price: 2.5 },
];

export default function App() {
  return (
    <main>
      <header>
        MiniShop <span aria-label="Items in cart">0</span>
      </header>
      {PRODUCTS.map((p) => (
        <button key={p.id}>Add {p.name}</button>
      ))}
      <ul aria-label="Cart"></ul>
      <p>Total: $0.00</p>
    </main>
  );
}
```

```jsx tests file=/App.test.js
import { fireEvent, render, screen, within } from "@testing-library/react";
import "@testing-library/jest-dom";
import App from "./App";

const add = (name) => fireEvent.click(screen.getByRole("button", { name: `Add ${name}` }));

test("adding the same product twice increases its quantity", () => {
  render(<App />);
  add("Coffee");
  add("Coffee");
  expect(within(screen.getByRole("list", { name: "Cart" })).getAllByRole("listitem")).toHaveLength(1);
  expect(screen.getByLabelText("Items in cart")).toHaveTextContent("2");
  expect(screen.getByText("Total: $8.00")).toBeInTheDocument();
});

test("totals several products with decimals", () => {
  render(<App />);
  add("Coffee");
  add("Tea");
  expect(screen.getByText("Total: $6.50")).toBeInTheDocument();
});

test("remove takes out only that line", () => {
  render(<App />);
  add("Coffee");
  add("Tea");
  fireEvent.click(screen.getByRole("button", { name: "Remove Coffee" }));
  expect(screen.getByLabelText("Items in cart")).toHaveTextContent("1");
  expect(screen.getByText("Total: $2.50")).toBeInTheDocument();
});
```

```jsx solution file=/App.js
import { useReducer } from "react";

const PRODUCTS = [
  { id: 1, name: "Coffee", price: 4 },
  { id: 2, name: "Tea", price: 2.5 },
];

function cartReducer(cart, action) {
  switch (action.type) {
    case "added": {
      const line = cart.find((l) => l.id === action.product.id);
      if (line) return cart.map((l) => (l === line ? { ...l, qty: l.qty + 1 } : l));
      return [...cart, { ...action.product, qty: 1 }];
    }
    case "removed":
      return cart.filter((l) => l.id !== action.id);
    default:
      return cart;
  }
}

export default function App() {
  const [cart, dispatch] = useReducer(cartReducer, []);
  const count = cart.reduce((n, l) => n + l.qty, 0);
  const total = cart.reduce((s, l) => s + l.price * l.qty, 0);

  return (
    <main>
      <header>
        MiniShop <span aria-label="Items in cart">{count}</span>
      </header>
      {PRODUCTS.map((p) => (
        <button key={p.id} onClick={() => dispatch({ type: "added", product: p })}>
          Add {p.name}
        </button>
      ))}
      <ul aria-label="Cart">
        {cart.map((l) => (
          <li key={l.id}>
            {l.name} × {l.qty}{" "}
            <button onClick={() => dispatch({ type: "removed", id: l.id })}>Remove {l.name}</button>
          </li>
        ))}
      </ul>
      <p>Total: ${total.toFixed(2)}</p>
    </main>
  );
}
```

`QtyStepper` shows a quantity between **1** and **10**. **+** and **−** change it by one and call `onChange(next)`; at the limits the button is disabled. Write tests that catch every buggy version.

```jsx code file=/QtyStepper.js
import { useState } from "react";

export default function QtyStepper({ initial = 1, onChange }) {
  const [qty, setQty] = useState(initial);
  function change(next) {
    setQty(next);
    onChange(next);
  }
  return (
    <div>
      <button aria-label="Decrease" disabled={qty <= 1} onClick={() => change(qty - 1)}>−</button>
      <span aria-label="Quantity">{qty}</span>
      <button aria-label="Increase" disabled={qty >= 10} onClick={() => change(qty + 1)}>+</button>
    </div>
  );
}
```

import { fireEvent, render, screen } from "@testing-library/react";
import "@testing-library/jest-dom";
import { expect, test, vi } from "vitest";
import QtyStepper from "./QtyStepper";

test("+ increases the quantity and reports the new value", () => {
  const onChange = vi.fn();
  render(<QtyStepper onChange={onChange} />);
  fireEvent.click(screen.getByLabelText("Increase"));
  expect(screen.getByLabelText("Quantity")).toHaveTextContent("2");
  expect(onChange).toHaveBeenCalledWith(2);
});

test("can't go below 1", () => {
  render(<QtyStepper onChange={vi.fn()} />);
  expect(screen.getByLabelText("Decrease")).toBeDisabled();
});

test("can't go above 10", () => {
  render(<QtyStepper initial={10} onChange={vi.fn()} />);
  expect(screen.getByLabelText("Increase")).toBeDisabled();
});

  hints={[

`statusLabel(status)` turns an order status into text for the customer, and `canCancel(status)` says whether the order can still be cancelled (only before it ships). Unknown statuses throw.

```js code file=orders.js
const LABELS = { placed: "Order placed", paid: "Payment received", shipped: "On its way", delivered: "Delivered" };

function statusLabel(status) {
  if (!(status in LABELS)) throw new Error("Unknown status: " + status);
  return LABELS[status];
}

function canCancel(status) {
  return status === "placed" || status === "paid";
}
```

```js starter
test("labels a placed order", () => {
  expect(statusLabel("placed")).toBe("Order placed");
});
```
